        function formatDate(dateStr) {
            if (!dateStr) return '';
            const parts = dateStr.split('-');
            if (parts.length === 3) {
                return `${parts[2]}/${parts[1]}/${parts[0]}`;
            }
            return dateStr;
        }

        let customers = JSON.parse(localStorage.getItem('kabir_customers')) || [];

        // Automatic Backup on page refresh / load
        function triggerRefreshBackup() {
            if (customers.length === 0) return;
            const blob = new Blob([JSON.stringify(customers, null, 2)], { type: 'application/json' });
            const url = window.URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.setAttribute('href', url);
            a.setAttribute('download', 'kabir_emporium_backup.json');
            document.body.appendChild(a);
            a.click();
            document.body.removeChild(a);
            window.URL.revokeObjectURL(url);
            showToast("Auto-backed up to 'kabir_emporium_backup.json'!");
        }

        function showToast(message, type = 'success') {
            const toast = document.getElementById('toast');
            const msgSpan = document.getElementById('toastMsg');
            const iconSpan = document.getElementById('toastIcon');
            
            msgSpan.innerText = message;
            iconSpan.innerHTML = type === 'success' ? '&#10003;' : '&#9888;';
            iconSpan.className = type === 'success' ? 'text-emerald-400 font-bold' : 'text-amber-400 font-bold';
            
            toast.classList.remove('translate-y-28', 'opacity-0');
            setTimeout(() => {
                toast.classList.add('translate-y-28', 'opacity-0');
            }, 3500);
        }

        function saveToStorage() {
            localStorage.setItem('kabir_customers', JSON.stringify(customers));
            renderTable();
        }

        function addCustomer(e) {
            e.preventDefault();
            const customer = {
                id: Date.now(),
                name: document.getElementById('name').value.trim(),
                city: document.getElementById('city').value.trim(),
                itemCount: parseInt(document.getElementById('itemCount').value),
                price: parseFloat(document.getElementById('price').value),
                date: document.getElementById('date').value
            };
            customers.unshift(customer);
            saveToStorage();
            document.getElementById('customerForm').reset();
            document.getElementById('date').valueAsDate = new Date();
            showToast('Customer record saved successfully!');
        }

        function deleteCustomer(id) {
            if(confirm('Are you sure you want to delete this customer record?')) {
                customers = customers.filter(c => c.id !== id);
                saveToStorage();
                showToast('Record deleted successfully!', 'warning');
            }
        }

        function renderTable() {
            const query = document.getElementById('searchCity').value.toLowerCase();
            const tbody = document.getElementById('customerTableBody');
            const mobileContainer = document.getElementById('mobileCardContainer');
            
            tbody.innerHTML = '';
            mobileContainer.innerHTML = '';

            let filtered = customers.filter(c =>
                c.city.toLowerCase().includes(query) || c.name.toLowerCase().includes(query)
            );

            let totalRev = 0;
            let totalItemsCount = 0;
            const uniqueCities = new Set();

            customers.forEach(c => {
                totalRev += (c.itemCount * c.price);
                totalItemsCount += c.itemCount;
                if(c.city) uniqueCities.add(c.city.trim().toLowerCase());
            });

            document.getElementById('totalRevenue').innerText = totalRev.toLocaleString('en-IN', {minimumFractionDigits: 2, maximumFractionDigits: 2});
            document.getElementById('recordCount').innerText = customers.length;

                                ${escapeHtml(c.city)}
                            </span>
                        </div>
                        <span class="text-xs font-medium text-slate-400">${formattedDate}</span>
                    </div>

                            <span class="font-medium text-slate-900">${c.itemCount}</span> items &times; ₹${c.price.toFixed(2)}

                        <div class="text-right">

                mobileContainer.appendChild(card);

                tbody.appendChild(tr);
            });
        }

        function escapeHtml(text) {
            const map = { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#039;' };
            return text.replace(/[&<>"']/g, function(m) { return map[m]; });
        }

        function backupData() {
            if(customers.length === 0) {
                showToast('No data available to backup!', 'error');
                return;
            }
            const blob = new Blob([JSON.stringify(customers, null, 2)], { type: 'application/json' });
            const url = window.URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.setAttribute('href', url);
            a.setAttribute('download', 'kabir_emporium_backup.json');
            a.click();
            showToast("JSON backup downloaded as 'kabir_emporium_backup.json'!");
        }

        function restoreData(event) {
            const file = event.target.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = function(e) {
                try {
                    const importedData = JSON.parse(e.target.result);
                    if (Array.isArray(importedData)) {
                        if (confirm(`Are you sure you want to restore ${importedData.length} records? This will replace your current records.`)) {
                            customers = importedData;
                            saveToStorage();
                            showToast(`Successfully restored ${importedData.length} records!`);
                        }
                    } else {
                        showToast('Invalid backup file format.', 'error');
                    }
                } catch(err) {
                    showToast('Failed to parse JSON backup file.', 'error');
                }
                event.target.value = '';
            };
            reader.readAsText(file);
        }

        renderTable();
        // Automatically trigger backup using the single filename on refresh / load
        triggerRefreshBackup();
